def transform(grid):
    H = len(grid)
    W = len(grid[0])
    bg = 0
    best = None
    for r in range(H):
        row = grid[r]
        if sum(1 for v in row if v == bg) * 2 > W:
            if best is None or r < best[0]:
                best = (r, row)
    if best is None:
        return grid
    br, brow = best
    holes = [c for c in range(W) if brow[c] == bg]
    wall = brow[holes[0]] if holes else 0

    boxes = {}
    seen = set()
    for r in range(H):
        for c in range(W):
            v = grid[r][c]
            if v == 0 or v == wall or (r, c) in seen:
                continue
            stack = [(r, c)]
            cells = []
            seen.add((r, c))
            while stack:
                y, x = stack.pop()
                cells.append((y, x))
                for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < H and 0 <= nx < W and grid[ny][nx] == v and (ny, nx) not in seen:
                        seen.add((ny, nx))
                        stack.append((ny, nx))
            boxes[v] = (cells, min(y for y, _ in cells), max(y for y, _ in cells), min(x for _, x in cells), max(x for _, x in cells))

    def shape_cells(cells):
        y0 = min(y for y, _ in cells)
        x0 = min(x for _, x in cells)
        return frozenset((y - y0, x - x0) for y, x in cells)

    # Find the three component shapes that combine to the demo pattern:
    # top bar, left/right cornered middle row, full middle row.
    target_top = frozenset((0, 0), (0, 1), (0, 2), (0, 3), (0, 4))
    target_mid = frozenset((1, 0), (1, 2), (1, 3), (1, 4))
    target_full = frozenset((2, 0), (2, 1), (2, 2), (2, 3), (2, 4))

    candidates = []
    for v, (cells, y0, y1, x0, x1) in boxes.items():
        sh = shape_cells(cells)
        if sh == target_top or sh == target_mid or sh == target_full:
            candidates.append((v, sh))

    def place_color(v, top, left):
        cells, _, _, _, _ = boxes[v]
        y0 = min(y for y, _ in cells)
        x0 = min(x for _, x in cells)
        for (y, x) in cells:
            ny, nx = top + (y - y0), left + (x - x0)
            if 0 <= ny < H and 0 <= nx < W and out[ny][nx] == 0:
                out[ny][nx] = v

    # Preserve the wall row and everything not moved, then paste selected components.
    out = [row[:] for row in grid]
    for v, (cells, y0, y1, x0, x1) in list(boxes.items()):
        for y, x in cells:
            out[y][x] = 0
        out[br] = list(brow)

    # Choose the anchor of the wall holes.
    if holes:
        ac = holes[0]
    else:
        ac = 0

    # Prefer the candidate whose original position is closest to being below the wall
    # and horizontally overlapping the left hole anchor.
    if candidates:
        def key(item):
            v, sh = item
            _, y0, y1, x0, x1 = boxes[v]
            # distance from bottom of shape to wall row, plus horizontal distance
            return (abs((y1 + 1) - br), abs(x0 - ac), v)
        candidates.sort(key=key)
        for v, sh in candidates:
            if sh == target_top:
                place_color(v, br - 1, ac - 1)
            elif sh == target_mid:
                place_color(v, br - 2, ac - 1)
            elif sh == target_full:
                place_color(v, br - 3, ac - 1)

    return out
