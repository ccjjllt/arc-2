def transform(grid):
    h = len(grid)
    w = len(grid[0])
    min_r = min_c = h * w
    max_r = max_c = -1
    for i, row in enumerate(grid):
        for j, v in enumerate(row):
            if v == 8:
                if i < min_r:
                    min_r = i
                if i > max_r:
                    max_r = i
                if j < min_c:
                    min_c = j
                if j > max_c:
                    max_c = j
    if min_c == h * w:
        return [[0]]

    bh = max_r - min_r + 1
    bw = max_c - min_c + 1
    best = None
    best_score = -1
    for dr in range(-h, h + 1):
        for dc in range(-w, w + 1):
            sr = min_r + dr
            sc = min_c + dc
            if sr < 0 or sc < 0 or sr + bh > h or sc + bw > w:
                continue
            score = 0
            ok = True
            for i in range(bh):
                r1 = min_r + i
                r2 = sr + i
                for j in range(bw):
                    c1 = min_c + j
                    c2 = sc + j
                    if grid[r2][c2] == 8:
                        if grid[r1][c1] != 8:
                            ok = False
                            break
                        continue
                    if grid[r1][c1] == 8:
                        continue
                    if grid[r1][c1] == grid[r2][c2]:
                        score += 1
                    else:
                        ok = False
                        break
                if not ok:
                    break
            if ok and score > best_score:
                best_score = score
                best = (sr, sc)

    if best is None:
        return [[0]]

    sr, sc = best
    filled = [[0] * bw for _ in range(bh)]
    for i in range(bh):
        for j in range(bw):
            v = grid[sr + i][sc + j]
            if v == 8:
                v = grid[min_r + i][min_c + j]
            filled[i][j] = v
    return filled
