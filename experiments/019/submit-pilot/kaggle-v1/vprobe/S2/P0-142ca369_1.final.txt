def transform(grid):
    H = len(grid)
    W = len(grid[0])
    out = [row[:] for row in grid]

    def is_empty(r, c):
        return 0 <= r < H and 0 <= c < W and grid[r][c] == 0

    def paint(r, c, dr, dc, val):
        while 0 <= r < H and 0 <= c < W and out[r][c] == 0:
            out[r][c] = val
            r += dr
            c += dc

    # straight three-cell segments
    for r in range(H - 2):
        for c in range(W):
            v = grid[r][c]
            if v and grid[r][c + 1] == v and grid[r][c + 2] == v:
                if c + 3 < W and grid[r][c + 3] == 0 and r + 1 < H and grid[r + 1][c] == 0:
                    paint(r + 1, c + 1, 1, 1, v)
                if c - 1 >= 0 and grid[r][c - 1] == 0 and r - 1 >= 0 and grid[r - 1][c] == 0:
                    paint(r - 1, c - 1, -1, -1, v)
            if v and grid[r + 1][c] == v and grid[r + 2][c] == v:
                if c + 1 < W and grid[r + 1][c + 1] == 0 and r + 3 < H and grid[r + 3][c] == 0:
                    paint(r + 2, c + 1, 1, 1, v)
                if c - 1 >= 0 and grid[r + 1][c - 1] == 0 and r - 1 >= 0 and grid[r - 1][c] == 0:
                    paint(r, c - 1, -1, -1, v)

    # L-trominoes, classified by the missing cell in their 2x2 box
    for r in range(H - 1):
        for c in range(W - 1):
            vals = [grid[r][c], grid[r][c + 1], grid[r + 1][c], grid[r + 1][c + 1]]
            nonzero = [v for v in vals if v]
            if len(nonzero) != 3 or len(set(nonzero)) != 1:
                continue
            v = nonzero[0]
            miss = vals.index(0)
            if miss == 0:  # top-left missing: cells (r,c+1),(r+1,c),(r+1,c+1)
                if is_empty(r + 2, c + 2):
                    paint(r + 2, c + 2, 1, 1, v)
                if is_empty(r - 1, c):
                    paint(r - 1, c, -1, -1, v)
            elif miss == 1:  # top-right missing
                if is_empty(r + 2, c):
                    paint(r + 2, c, 1, -1, v)
                if is_empty(r - 1, c + 1):
                    paint(r - 1, c + 1, -1, 1, v)
            elif miss == 2:  # bottom-left missing
                if is_empty(r, c - 1):
                    paint(r, c - 1, -1, -1, v)
                if is_empty(r + 1, c + 2):
                    paint(r + 1, c + 2, 1, 1, v)
            else:  # bottom-right missing
                if is_empty(r, c + 2):
                    paint(r, c + 2, 1, 1, v)
                if is_empty(r + 1, c - 1):
                    paint(r + 1, c - 1, 1, -1, v)

    return out
