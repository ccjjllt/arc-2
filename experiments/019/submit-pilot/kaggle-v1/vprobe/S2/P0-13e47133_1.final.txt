def transform(grid):
    H=len(grid); W=len(grid[0])
    def is_wall(g,r,c):
        return 0<=r<H and 0<=c<W and g[r][c] in (1,2,6)
    def flood(g,sr,sc):
        seen=[[False]*W for _ in range(H)]
        q=[(sr,sc)]; seen[sr][sc]=True; pts=[]
        while q:
            r,c=q.pop(); pts.append((r,c))
            for dr,dc in ((1,0),(-1,0),(0,1),(0,-1)):
                nr,nc=r+dr,c+dc
                if 0<=nr<H and 0<=nc<W and not seen[nr][nc] and not is_wall(g,nr,nc):
                    seen[nr][nc]=True; q.append((nr,nc))
        return pts
    out=[row[:] for row in grid]
    for i in range(H):
        for j in range(W):
            if grid[i][j] in (1,2,6):
                continue
            for cell in flood(grid,i,j):
                r,c=cell
                d=min(r+1,c+1,H-r,W-c)
                if d==1:
                    continue
                if out[r][c] in (1,2,6):
                    out[r][c]=grid[i][j]
    return out
