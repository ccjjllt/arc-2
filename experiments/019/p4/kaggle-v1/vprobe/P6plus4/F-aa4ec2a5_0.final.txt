def transform(grid):
    H = len(grid)
    W = len(grid[0])
    g = [row[:] for row in grid]
    out = [row[:] for row in grid]

    def inb(r, c):
        return 0 <= r < H and 0 <= c < W

    comps = []
    for r in range(H):
        for c in range(W):
            if g[r][c] == 1:
                stack = [(r, c)]
                g[r][c] = 0
                cells = []
                while stack:
                    x, y = stack.pop()
                    cells.append((x, y))
                    for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nx, ny = x + dx, y + dy
                        if inb(nx, ny) and g[nx][ny] == 1:
                            g[nx][ny] = 0
                            stack.append((nx, ny))
                comps.append(cells)

    for cells in comps:
        if not cells:
            continue
        s = set(cells)
        holes = []
        seen_hole = set()
        for r, c in s:
            for nr, nc in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if inb(nr, nc) and grid[nr][nc] == 4 and (nr, nc) not in seen_hole:
                    stack = [(nr, nc)]
                    seen_hole.add((nr, nc))
                    comp_bg = []
                    closed = True
                    while stack:
                        x, y = stack.pop()
                        comp_bg.append((x, y))
                        for nx, ny in ((x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)):
                            if not inb(nx, ny):
                                closed = False
                                continue
                            if (nx, ny) in seen_hole or grid[nx][ny] != 4:
                                continue
                            seen_hole.add((nx, ny))
                            stack.append((nx, ny))
                    if closed:
                        holes.extend(comp_bg)
        hole_set = set(holes)
        interior = s | hole_set
        for r, c in s:
            out[r][c] = 8 if hole_set else 1
        for r, c in holes:
            out[r][c] = 6
        border = set()
        for r, c in interior:
            for nr, nc in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if inb(nr, nc) and (nr, nc) not in interior and out[nr][nc] == 4:
                    border.add((nr, nc))
        for r, c in border:
            out[r][c] = 2
    return out
