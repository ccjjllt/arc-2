def transform(grid):
    H = len(grid)
    W = len(grid[0])
    bg = grid[0][0]
    out = [row[:] for row in grid]
    seen = [[False] * W for _ in range(H)]

    def inside(r, c):
        return 0 <= r < H and 0 <= c < W and not seen[r][c] and grid[r][c] != bg and grid[r][c] != 0

    for sr in range(H):
        for sc in range(W):
            if not inside(sr, sc):
                continue
            q = [(sr, sc)]
            seen[sr][sc] = True
            cells = [(sr, sc)]
            while q:
                r, c = q.pop()
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if inside(nr, nc):
                        seen[nr][nc] = True
                        q.append((nr, nc))
                        cells.append((nr, nc))
            if len(cells) == 1:
                r, c = cells[0]
                for dr in (-1, 1):
                    for dc in (-1, 1):
                        if 0 <= r + dr < H and 0 <= c + dc < W:
                            out[r + dr][c + dc] = grid[r][c]
    return out
