def transform(grid):
    if not grid or not grid[0]:
        return [[0]]

    H = len(grid)
    W = len(grid[0])

    zero = None
    for r in range(H):
        for c in range(W):
            if grid[r][c] == 0:
                zero = (r, c)
                break
        if zero is not None:
            break

    zr, zc = zero

    # Determine the inner region: the frame is the top row plus either the
    # left or right column.  All demos use a top corner zero.
    if zc == 0:
        r0, r1 = zr + 1, H
        c0, c1 = zc + 1, W
        horizontal_step = 1
    elif zc == W - 1:
        r0, r1 = zr + 1, H
        c0, c1 = 0, zc
        horizontal_step = -1
    else:
        # Fallback for an unexpected corner: treat the whole rectangle inside
        # the bounding box of non-8 cells connected to the zero as inner.
        seen = [[False] * W for _ in range(H)]
        stack = [zero]
        seen[zr][zc] = True
        while stack:
            r, c = stack.pop()
            for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nr, nc = r + dr, c + dc
                if 0 <= nr < H and 0 <= nc < W and not seen[nr][nc] and grid[nr][nc] != 8:
                    seen[nr][nc] = True
                    stack.append((nr, nc))
        fr = [r for r in range(H) for c in range(W) if seen[r][c]]
        fc = [c for r in range(H) for c in range(W) if seen[r][c]]
        r0, r1 = min(fr) + 1, max(fr)
        c0, c1 = min(fc) + 1, max(fc)
        horizontal_step = 1

    h = r1 - r0
    w = c1 - c0

    # Connected components of non-8 cells inside the inner region.
    visited = [[False] * w for _ in range(h)]
    objects = []

    for sr in range(h):
        for sc in range(w):
            if visited[sr][sc] or grid[r0 + sr][c0 + sc] == 8:
                continue

            stack = [(sr, sc)]
            visited[sr][sc] = True
            cells = []

            while stack:
                r, c = stack.pop()
                cells.append((r, c))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if (0 <= nr < h and 0 <= nc < w and not visited[nr][nc]
                            and grid[r0 + nr][c0 + nc] != 8):
                        visited[nr][nc] = True
                        stack.append((nr, nc))

            rmin = min(r for r, _ in cells)
            rmax = max(r for r, _ in cells) + 1
            cmin = min(c for _, c in cells)
            cmax = max(c for _, c in cells) + 1

            mat = []
            for rr in range(rmin, rmax):
                row = []
                for cc in range(cmin, cmax):
                    val = grid[r0 + rr][c0 + cc]
                    if val == 8:
                        val = 0
                    row.append(val)
                mat.append(row)

            objects.append((rmin, cmin, rmax - rmin, cmax - cmin, mat))

    # Reading order:
    # - primary direction follows the longer inner side.
    # - horizontal order moves away from the zero corner.
    # - vertical order is always top to bottom.
    if h > w:
        if horizontal_step > 0:
            col_keys = sorted(set(o[1] for o in objects))
        else:
            col_keys = sorted(set(o[1] for o in objects), reverse=True)

        ordered = []
        for c in col_keys:
            in_col = [o for o in objects if o[1] == c]
            in_col.sort(key=lambda o: o[0])
            ordered.extend(in_col)
    else:
        row_keys = sorted(set(o[0] for o in objects))
        ordered = []
        for r in row_keys:
            in_row = [o for o in objects if o[0] == r]
            if horizontal_step > 0:
                in_row.sort(key=lambda o: o[1])
            else:
                in_row.sort(key=lambda o: o[1], reverse=True)
            ordered.extend(in_row)

    # Rotate tall objects 90 degrees counter-clockwise so their width is
    # at least their height.
    pieces = []
    for _, _, hh, ww, mat in ordered:
        if hh > ww:
            new_h = ww
            new_w = hh
            rotated = [[0] * new_w for _ in range(new_h)]
            for i in range(hh):
                for j in range(ww):
                    rotated[j][i] = mat[i][ww - 1 - j]
            pieces.append(rotated)
        else:
            pieces.append(mat)

    if not pieces:
        return [[8]]

    out_w = max(max(len(row) for row in p) for p in pieces)
    out = []
    for p in pieces:
        ph = len(p)
        pw = len(p[0])
        left = (out_w - pw) // 2
        for row in p:
            out.append([8] * left + row + [8] * (out_w - pw - left))

    return out
